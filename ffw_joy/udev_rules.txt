# ==============================================================================
# ffw_joy UDEV Rules — Persistent Device Symlinks
# ==============================================================================
# Applies to: 3Dconnexion SpaceMouse + Logitech G Extreme 3D Pro
#
# 1. Copy this file (after editing phys/serial match values) to:
#      sudo cp src/ai_worker/ffw_joy/udev_rules.txt /etc/udev/rules.d/99-ffw-joysticks.rules
# 2. Reload the rules:
#      sudo udevadm control --reload-rules && sudo udevadm trigger
#
# To discover your own device's phys or serial values, run:
#   python3 src/ai_worker/ffw_joy/scripts/identify_joysticks.py
# ==============================================================================

# ──────────────────────────────────────────────
# 3Dconnexion SpaceMouse — LEFT ARM
# Bind to physical USB port 1.4.4
# ──────────────────────────────────────────────
KERNEL=="js*", SUBSYSTEM=="input", ATTRS{phys}=="usb-0000:00:14.0-1.4.4/input0", SYMLINK+="input/spacemouse_left"

# ──────────────────────────────────────────────
# 3Dconnexion SpaceMouse — RIGHT ARM (or dongle)
# Bind to physical USB port 1.1.2
# ──────────────────────────────────────────────
KERNEL=="js*", SUBSYSTEM=="input", ATTRS{phys}=="usb-0000:00:14.0-1.1.2/input0", SYMLINK+="input/spacemouse_right"

# ──────────────────────────────────────────────
# Logitech G Extreme 3D Pro
# Vendor ID 046d, Product ID c215
# ──────────────────────────────────────────────
# Option A: Bind by product ID (works for a single Logitech joystick)
KERNEL=="js*", SUBSYSTEM=="input", ATTRS{idVendor}=="046d", ATTRS{idProduct}=="c215", SYMLINK+="input/logitech_3dpro"
