# Template UDEV Rules for Binding Joystick Devices to Static Names
# Save this file (after editing) to: /etc/udev/rules.d/99-ffw-joysticks.rules
# Then run: sudo udevadm control --reload-rules && sudo udevadm trigger
#
# Supported devices:
#   - 3Dconnexion SpaceMouse (vendor 256f)
#   - Logitech G Extreme 3D Pro (vendor 046d, product c215)
#
# --------------------------------------------------------------------------------
# HOW TO FIND YOUR DEVICE IDENTIFIER:
# 1. Plug in your device.
# 2. Run: python3 src/ai_worker/ffw_joy/scripts/identify_joysticks.py
# 3. Use the printed phys, serial, or idProduct/idVendor values to build rules.
# --------------------------------------------------------------------------------

# ──────────────────────────────────────────────
# Example 1: Bind SpaceMouse by physical USB port
# ──────────────────────────────────────────────
KERNEL=="js*", SUBSYSTEM=="input", ATTRS{phys}=="usb-0000:00:14.0-1.4.4/input0", SYMLINK+="input/spacemouse_left"
KERNEL=="js*", SUBSYSTEM=="input", ATTRS{phys}=="usb-0000:00:14.0-1.1.2/input0", SYMLINK+="input/spacemouse_right"

# ──────────────────────────────────────────────
# Example 2: Bind SpaceMouse by serial number
# ──────────────────────────────────────────────
# KERNEL=="js*", SUBSYSTEM=="input", ATTRS{serial}=="12345678", SYMLINK+="input/spacemouse_left"
# KERNEL=="js*", SUBSYSTEM=="input", ATTRS{serial}=="87654321", SYMLINK+="input/spacemouse_right"
